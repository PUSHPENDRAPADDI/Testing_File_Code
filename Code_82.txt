### Step 5: Implement FastAPI Endpoints

#### `main.py`

This file will implement the FastAPI application with the necessary endpoints.

```python
from fastapi import FastAPI, HTTPException, Depends
from sqlalchemy.orm import Session
from . import models, schemas
from .database import SessionLocal, engine

models.Base.metadata.create_all(bind=engine)

app = FastAPI()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.post("/claims/", response_model=schemas.ClaimResponse)
def create_claim(claim: schemas.ClaimCreate, db: Session = Depends(get_db)):
    # Validate and match patient and provider
    patient = db.query(models.Patient).filter(
        models.Patient.id == claim.patient_id).first()
    provider = db.query(models.Provider).filter(
        models.Provider.id == claim.provider_id).first()
    
    if not patient or not provider:
        raise HTTPException(status_code=404, detail="Patient or Provider not found")

    # Reprice claim (simple logic for demonstration)
    repriced_amount = claim.amount * 0.8  # Apply a discount for demonstration

    db_claim = models.Claim(**claim.dict(), amount=repriced_amount, status="Processed")
    db.add(db_claim)
    db.commit()
    db.refresh(db_claim)
    
    return db_claim

@app.get("/claims/{claim_id}", response_model=schemas.ClaimResponse)
def read_claim(claim_id: int, db: Session = Depends(get_db)):
    claim = db.query(models.Claim).filter(models.Claim.id == claim_id).first()
    if claim is None:
        raise HTTPException(status_code=404, detail="Claim not found")
    return claim
```
