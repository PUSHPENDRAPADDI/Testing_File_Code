```python
from fastapi import FastAPI, HTTPException, UploadFile, File
from pydantic import BaseModel
from typing import List, Optional
import xml.etree.ElementTree as ET

app = FastAPI()

# Model for Claim
class Claim(BaseModel):
    patient_first_name: str
    patient_last_name: str
    patient_dob: str
    provider_tin: str
    provider_npi: str
    provider_last_name: str
    claim_amount: float

class ClaimResponse(BaseModel):
    claim_id: str
    status: str
    repriced_amount: Optional[float] = None
    error_message: Optional[str] = None

# Mock database
patients_db = []  # This would be a database in a real application.
providers_db = []  # This would be a database in a real application.

@app.post("/import-claim", response_model=ClaimResponse)
async def import_claim(claim: Claim):
    # Validate patient
    patient_record = next((p for p in patients_db if (p['first_name'] == claim.patient_first_name and
                                                       p['last_name'] == claim.patient_last_name and
                                                       p['dob'] == claim.patient_dob)), None)
    if not patient_record:
        return ClaimResponse(claim_id="N/A", status="Failed", error_message="Patient not found")

    # Validate provider
    provider_record = next((p for p in providers_db if (p['tin'] == claim.provider_tin and
                                                         p['npi'] == claim.provider_npi and
                                                         p['last_name'] == claim.provider_last_name)), None)
    if not provider_record:
        return ClaimResponse(claim_id="N/A", status="Failed", error_message="Provider not found")

    # Reprice the claim (mock repricing logic)
    repriced_amount = claim.claim_amount * 0.8  # Example repricing logic

    # Here you would add logic to save the claim and return a response
    return ClaimResponse(claim_id="1234", status="Success", repriced_amount=repriced_amount)

@app.post("/upload-claims", response_model=List[ClaimResponse])
async def upload_claims(file: UploadFile = File(...)):
    if file.content_type != 'application/xml':
        raise HTTPException(status_code=400, detail="Invalid file type. Only XML files are accepted")

    # Parse XML file
    try:
        tree = ET.parse(file.file)
        root = tree.getroot()
        claims = []
        
        for claim_elem in root.findall('claim'):
            claim_data = Claim(
                patient_first_name=claim_elem.find('patient_first_name').text,
                patient_last_name=claim_elem.find('patient_last_name').text,
                patient_dob=claim_elem.find('patient_dob').text,
                provider_tin=claim_elem.find('provider_tin').text,
                provider_npi=claim_elem.find('provider_npi').text,
                provider_last_name=claim_elem.find('provider_last_name').text,
                claim_amount=float(claim_elem.find('claim_amount').text)
            )
            claims.append(claim_data)

        responses = []
        for claim in claims:
            response = await import_claim(claim)
            responses.append(response)

        return responses

    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Error processing file: {str(e)}")

# Mock data for testing
patients_db.append({'first_name': 'John', 'last_name': 'Doe', 'dob': '1980-01-01'})
providers_db.append({'tin': '123456789', 'npi': '987654321', 'last_name': 'Smith'})

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host='0.0.0.0', port=8000)
```

### Explanation:

- **Models**: We define a `Claim` model for incoming claims and `ClaimResponse` for the outgoing responses.
- **Endpoints**:
  - `POST /import-claim`: Receives a single claim, validates it against mock databases, reprices it, and returns a response.
  - `POST /upload-claims`: Accepts an XML file, parses it, and imports multiple claims using the `import_claim` function.
- **Mock Databases**: For demonstration, we use simple lists to store patients and providers. In a real implementation, you would connect to a PostgreSQL database to validate and store claims.
