def validate_claim(claim: Claim) -> Claim:
    # Validate incoming claim data
    if not claim.patient_first_name or not claim.patient_last_name:
        raise HTTPException(status_code=400, detail="Invalid patient data")
    if not claim.provider_tin or not claim.provider_npi:
        raise HTTPException(status_code=400, detail="Invalid provider data")

    for patient in patients_db:

    for provider in providers_db:
        if (provider["tin"] == claim.provider_tin and
            provider["npi"] == claim.provider_npi and
