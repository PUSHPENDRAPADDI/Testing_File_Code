      console.log('Saved Profile Data', this.clientProfileForm.value);
      this.status = 'complete';
      this.updateStatusAndCompletion();

  updateStatusAndCompletion(): void {
    const totalFields = Object.keys(this.clientProfileForm.controls).length;
    const filledFields = Object.keys(this.clientProfileForm.controls).filter(field => 
      this.clientProfileForm.get(field)?.value).length;

    this.percentageCompletion = (filledFields / totalFields) * 100;

    if (this.percentageCompletion === 0) {
      this.status = 'not_started';
    } else if (this.percentageCompletion < 100) {
      this.status = 'in_process';
    } else {
      this.status = 'complete';
    }
