### Code Implementation

```javascript
// Import necessary libraries
const express = require('express');
const mongoose = require('mongoose');
const bodyParser = require('body-parser');

// Initialize express app
const app = express();
const PORT = process.env.PORT || 3000;

// Middleware
app.use(bodyParser.json());

// MongoDB Client Profile Schema
const clientProfileSchema = new mongoose.Schema({
    name: { type: String, required: true },
    email: { type: String, required: true, unique: true },
    phone: { type: String, required: true }
});

// Mongoose model
const ClientProfile = mongoose.model('ClientProfile', clientProfileSchema);

// Connect to MongoDB
mongoose.connect('mongodb://localhost:27017/clientDB', {
    useNewUrlParser: true,
    useUnifiedTopology: true
});

// Create a Client Profile
app.post('/api/client-profiles', async (req, res) => {
    try {
        const clientProfile = new ClientProfile(req.body);
        await clientProfile.save();
        res.status(201).json(clientProfile);
    } catch (error) {
        res.status(400).json({ message: error.message });
    }
});

// View Client Profile
app.get('/api/client-profiles/:id', async (req, res) => {
    try {
        const clientProfile = await ClientProfile.findById(req.params.id);
        if (!clientProfile) return res.status(404).json({ message: "Profile not found" });
        res.status(200).json(clientProfile);
    } catch (error) {
        res.status(500).json({ message: error.message });
    }
});

// Update Client Profile
app.put('/api/client-profiles/:id', async (req, res) => {
    try {
        const clientProfile = await ClientProfile.findByIdAndUpdate(req.params.id, req.body, { new: true, runValidators: true });
        if (!clientProfile) return res.status(404).json({ message: "Profile not found" });
        res.status(200).json(clientProfile);
    } catch (error) {
        res.status(400).json({ message: error.message });
    }
});

// Delete Client Profile
app.delete('/api/client-profiles/:id', async (req, res) => {
    try {
        const clientProfile = await ClientProfile.findByIdAndDelete(req.params.id);
        if (!clientProfile) return res.status(404).json({ message: "Profile not found" });
        res.status(204).send();
    } catch (error) {
        res.status(500).json({ message: error.message });
    }
});

// Start the server
app.listen(PORT, () => {
    console.log(`Server is running on http://localhost:${PORT}`);
});
```
