### 2. Update Product Controller for ClientProfile

You will need to modify the controller to handle CRUD operations for the new `ClientProfile` model.

```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly ICustomService<ClientProfile> _customService;

        public ClientProfileController(ICustomService<ClientProfile> customService)
        {
            _customService = customService;
        }

        [HttpGet(nameof(GetProfileById))]
        public IActionResult GetProfileById(int Id)
        {
            var profile = _customService.Get(Id);
            if (profile == null)
            {
                return NotFound();
            }
            return Ok(profile);
        }

        [HttpGet(nameof(GetAllProfiles))]
        public IActionResult GetAllProfiles()
        {
            var profiles = _customService.GetAll();
            return Ok(profiles);
        }

        [HttpPost(nameof(CreateProfile))]
        public IActionResult CreateProfile(ClientProfile clientProfile)
        {
            if (clientProfile != null)
            {
                _customService.Insert(clientProfile);
                return Ok("Profile created successfully.");
            }
            return BadRequest("Invalid client profile data.");
        }

        [HttpPut(nameof(UpdateProfile))]
        public IActionResult UpdateProfile(ClientProfile clientProfile)
        {
            if (clientProfile != null)
            {
                _customService.Update(clientProfile);
                return Ok("Profile updated successfully.");
            }
            return BadRequest("Invalid client profile data.");
        }

        [HttpDelete(nameof(DeleteProfile))]
        public IActionResult DeleteProfile(int Id)
        {
            var profile = _customService.Get(Id);
            if (profile != null)
            {
                _customService.Delete(profile);
                return Ok("Profile deleted successfully.");
            }
            return NotFound("Profile not found.");
        }
    }
}
```
