Certainly! Below is a FastAPI implementation for the Healthcare API System based on your Business Requirements Document (BRD). The structure includes endpoints for Patient Management, Appointment Scheduling, Electronic Medical Records (EMR), Prescription Management, and Billing & Insurance.

### FastAPI Implementation

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional

app = FastAPI()

# Mock databases
patients_db = {}
appointments_db = {}
records_db = {}
prescriptions_db = {}
bills_db = {}

# Models
class Patient(BaseModel):
    id: int
    name: str
    contact_details: str
    medical_history: str

class Appointment(BaseModel):
    id: int
    patient_id: int
    doctor_id: int
    datetime: str

class MedicalRecord(BaseModel):
    patient_id: int
    record: str

class Prescription(BaseModel):
    id: int
    patient_id: int
    doctor_id: int
    medication: str

class Bill(BaseModel):
    id: int
    patient_id: int
    amount: float
    status: str

# Patient Management API
@app.post("/patients", response_model=Patient)
async def register_patient(patient: Patient):
    if patient.id in patients_db:
        raise HTTPException(status_code=400, detail="Patient already exists")
    patients_db[patient.id] = patient
    return patient

@app.get("/patients/{patient_id}", response_model=Patient)
async def get_patient(patient_id: int):
    if patient_id not in patients_db:
        raise HTTPException(status_code=404, detail="Patient not found")
    return patients_db[patient_id]

@app.put("/patients/{patient_id}", response_model=Patient)
async def update_patient(patient_id: int, patient: Patient):
    if patient_id not in patients_db:
        raise HTTPException(status_code=404, detail="Patient not found")
    patients_db[patient_id] = patient
    return patient

@app.delete("/patients/{patient_id}", response_model=dict)
async def deactivate_patient(patient_id: int):
    if patient_id not in patients_db:
        raise HTTPException(status_code=404, detail="Patient not found")
    del patients_db[patient_id]
    return {"detail": "Patient deactivated"}

# Appointment Scheduling API
@app.post("/appointments", response_model=Appointment)
async def schedule_appointment(appointment: Appointment):
    if appointment.id in appointments_db:
        raise HTTPException(status_code=400, detail="Appointment already exists")
    appointments_db[appointment.id] = appointment
    return appointment

@app.put("/appointments/{appointment_id}", response_model=Appointment)
async def modify_appointment(appointment_id: int, appointment: Appointment):
    if appointment_id not in appointments_db:
        raise HTTPException(status_code=404, detail="Appointment not found")
    appointments_db[appointment_id] = appointment
    return appointment

@app.get("/doctors/{doctor_id}/availability")
async def get_doctor_availability(doctor_id: int):
    # Mocked availability for simplicity
    return {"doctor_id": doctor_id, "available": True}

# EMR API
@app.get("/patients/{patient_id}/records", response_model=List[MedicalRecord])
async def get_medical_records(patient_id: int):
    return [record for record in records_db.get(patient_id, [])]

@app.post("/patients/{patient_id}/records", response_model=MedicalRecord)
async def add_medical_record(patient_id: int, record: MedicalRecord):
    if patient_id not in records_db:
        records_db[patient_id] = []
    records_db[patient_id].append(record)
    return record

# Prescription Management API
@app.post("/prescriptions", response_model=Prescription)
async def generate_prescription(prescription: Prescription):
    if prescription.id in prescriptions_db:
        raise HTTPException(status_code=400, detail="Prescription already exists")
    prescriptions_db[prescription.id] = prescription
    return prescription

@app.get("/prescriptions/{prescription_id}", response_model=Prescription)
async def get_prescription(prescription_id: int):
    if prescription_id not in prescriptions_db:
        raise HTTPException(status_code=404, detail="Prescription not found")
    return prescriptions_db[prescription_id]

# Billing & Insurance API
@app.post("/billing", response_model=Bill)
async def generate_invoice(bill: Bill):
    if bill.id in bills_db:
        raise HTTPException(status_code=400, detail="Bill already exists")
    bills_db[bill.id] = bill
    return bill

@app.get("/billing/{bill_id}", response_model=Bill)
async def get_invoice(bill_id: int):
    if bill_id not in bills_db:
        raise HTTPException(status_code=404, detail="Bill not found")
    return bills_db[bill_id]

@app.post("/insurance/claims")
async def process_insurance_claim(bill: Bill):
    # Mock processing insurance claims
    return {"status": "Claim processed", "bill_id": bill.id}

# Run the application with: uvicorn filename:app --reload
```

### Instructions to Run the API
1. Save the above code into a file named `main.py`.
2. Install FastAPI and Uvicorn if you haven't already:
   ```bash
   pip install fastapi uvicorn
   ```
3. Run the FastAPI application:
   ```bash
   uvicorn main:app --reload
   ```
4. Open your browser and navigate to `http://127.0.0.1:8000/docs` to access the interactive API documentation.

### Note
This implementation uses in-memory dictionaries to simulate databases for demonstration purposes. In a production environment, you would typically use a database management system (DBMS) to store this data persistently. Moreover, the security features (like OAuth 2.0/JWT) and error handling can be further expanded based on specific requirements.
