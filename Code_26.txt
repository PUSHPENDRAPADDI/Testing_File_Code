### Step 3: Create the Repository

Now, create a repository to handle the operations for client profiles.

```csharp
using System.Collections.Generic;
using System.Linq;

public class ClientProfileRepository
{
    private readonly ApplicationDbContext _context;

    public ClientProfileRepository(ApplicationDbContext context)
    {
        _context = context;
    }

    public void AddClientProfile(ClientProfile profile)
    {
        profile.CreatedAt = DateTime.Now;
        profile.UpdatedAt = DateTime.Now;
        _context.ClientProfiles.Add(profile);
        _context.SaveChanges();
    }

    public ClientProfile GetClientProfile(int id)
    {
        return _context.ClientProfiles.Find(id);
    }

    public IEnumerable<ClientProfile> GetAllClientProfiles()
    {
        return _context.ClientProfiles.ToList();
    }

    public void UpdateClientProfile(ClientProfile profile)
    {
        profile.UpdatedAt = DateTime.Now;
        _context.ClientProfiles.Update(profile);
        _context.SaveChanges();
    }

    public void DeleteClientProfile(int id)
    {
        var profile = _context.ClientProfiles.Find(id);
        if (profile != null)
        {
            _context.ClientProfiles.Remove(profile);
            _context.SaveChanges();
        }
    }
}
```

### Step 5: Create the Controller

Finally, create a simple ASP.NET Core controller to expose these functionalities via an API.

```csharp
using Microsoft.AspNetCore.Mvc;

[ApiController]
[Route("api/[controller]")]
public class ClientProfilesController : ControllerBase
{
    private readonly ClientProfileService _service;

    public ClientProfilesController(ClientProfileService service)
    {
        _service = service;
    }

    [HttpPost]
    public IActionResult CreateProfile([FromBody] ClientProfile profile)
    {
        _service.CreateProfile(profile);
        return CreatedAtAction(nameof(GetProfile), new { id = profile.Id }, profile);
    }

    [HttpGet("{id}")]
    public IActionResult GetProfile(int id)
    {
        var profile = _service.GetProfile(id);
        if (profile == null)
        {
            return NotFound();
        }
        return Ok(profile);
    }

    [HttpGet]
    public IActionResult GetAllProfiles()
    {
        var profiles = _service.GetAllProfiles();
        return Ok(profiles);
    }

    [HttpPut("{id}")]
    public IActionResult UpdateProfile(int id, [FromBody] ClientProfile profile)
    {
        if (id != profile.Id)
        {
            return BadRequest();
        }
        _service.UpdateProfile(profile);
        return NoContent();
    }

    [HttpDelete("{id}")]
    public IActionResult DeleteProfile(int id)
    {
        _service.DeleteProfile(id);
        return NoContent();
    }
}
```
