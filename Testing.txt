To develop an Express.js-based application that fulfills the requirements specified in the Business Requirements Document (BRD) for a Healthcare API System, we will create a basic structure that includes the necessary API endpoints for each module: Patient Management, Appointment Scheduling, Electronic Medical Records (EMR), Prescription Management, and Billing & Insurance.

Below is a sample Express.js server setup with routes for the specified APIs. This code assumes you have Node.js and Express installed, along with other packages like `body-parser` for parsing request bodies, and `mongoose` if you plan to connect to a MongoDB database.

### Step 1: Install Required Packages

First, make sure you have `express`, `body-parser`, and any other necessary packages installed:

```bash
npm install express body-parser mongoose
```

### Step 2: Create the Express Server

Create a file named `server.js` and add the following code:

```javascript
const express = require('express');
const bodyParser = require('body-parser');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(bodyParser.json());

// Mock database
let patients = [];
let appointments = [];
let medicalRecords = [];
let prescriptions = [];
let invoices = [];

// Patient Management API
app.post('/patients', (req, res) => {
    const newPatient = req.body;
    patients.push(newPatient);
    res.status(201).json(newPatient);
});

app.get('/patients/:id', (req, res) => {
    const patient = patients.find(p => p.id === req.params.id);
    if (patient) {
        res.json(patient);
    } else {
        res.status(404).send('Patient not found');
    }
});

app.put('/patients/:id', (req, res) => {
    const index = patients.findIndex(p => p.id === req.params.id);
    if (index !== -1) {
        patients[index] = { ...patients[index], ...req.body };
        res.json(patients[index]);
    } else {
        res.status(404).send('Patient not found');
    }
});

app.delete('/patients/:id', (req, res) => {
    const index = patients.findIndex(p => p.id === req.params.id);
    if (index !== -1) {
        patients.splice(index, 1);
        res.status(204).send();
    } else {
        res.status(404).send('Patient not found');
    }
});

// Appointment Scheduling API
app.post('/appointments', (req, res) => {
    const newAppointment = req.body;
    appointments.push(newAppointment);
    res.status(201).json(newAppointment);
});

app.put('/appointments/:id', (req, res) => {
    const index = appointments.findIndex(a => a.id === req.params.id);
    if (index !== -1) {
        appointments[index] = { ...appointments[index], ...req.body };
        res.json(appointments[index]);
    } else {
        res.status(404).send('Appointment not found');
    }
});

// Retrieve doctor availability
app.get('/doctors/:id/availability', (req, res) => {
    // Mock availability data
    res.json({ available: true, slots: ['9:00 AM', '10:00 AM', '11:00 AM'] });
});

// EMR API
app.get('/patients/:id/records', (req, res) => {
    const records = medicalRecords.filter(record => record.patientId === req.params.id);
    res.json(records);
});

app.post('/patients/:id/records', (req, res) => {
    const newRecord = { ...req.body, patientId: req.params.id };
    medicalRecords.push(newRecord);
    res.status(201).json(newRecord);
});

// Prescription Management API
app.post('/prescriptions', (req, res) => {
    const newPrescription = req.body;
    prescriptions.push(newPrescription);
    res.status(201).json(newPrescription);
});

app.get('/prescriptions/:id', (req, res) => {
    const prescription = prescriptions.find(p => p.id === req.params.id);
    if (prescription) {
        res.json(prescription);
    } else {
        res.status(404).send('Prescription not found');
    }
});

// Billing & Insurance API
app.post('/billing', (req, res) => {
    const newInvoice = req.body;
    invoices.push(newInvoice);
    res.status(201).json(newInvoice);
});

app.get('/billing/:id', (req, res) => {
    const invoice = invoices.find(i => i.id === req.params.id);
    if (invoice) {
        res.json(invoice);
    } else {
        res.status(404).send('Invoice not found');
    }
});

// Process insurance claims
app.post('/insurance/claims', (req, res) => {
    // Mock processing insurance claims
    res.status(201).json({ status: 'Claim submitted', claimId: '12345' });
});

// Start the server
app.listen(PORT, () => {
    console.log(`Server is running on port ${PORT}`);
});
```

### Step 3: Run the Server

Run the server using the following command:

```bash
node server.js
```

### Step 4: Test the API

You can use tools like Postman or curl to test your API endpoints. Make sure to send requests with the appropriate JSON body where needed.
