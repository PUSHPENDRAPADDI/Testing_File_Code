import { Component, OnInit } from '@angular/core';
import { FormBuilder, FormGroup, Validators } from '@angular/forms';
import { ClientProfileService } from './client-profile.service';

@Component({
  selector: 'app-client-profile-form',
  templateUrl: './client-profile-form.component.html',

  races = ['White', 'Black or African', 'American Indian or Alaska Native', 'Asian', 'Native Hawaiian or other Pacific Islander', 'Hispanic or Latino', 'Prefer not to answer', 'Other'];

  constructor(private fb: FormBuilder, private clientProfileService: ClientProfileService) {
    this.clientProfileForm = this.fb.group({
      firstName: ['', [Validators.required, Validators.maxLength(50)]],
      lastName: ['', [Validators.required, Validators.maxLength(50)]],
      gender: [''],
      race: [[]],
      birthDate: ['', [Validators.required]],
      assignedTo: [{ value: '', disabled: true }], // System Generated
      // More fields based on your specifications
      primaryPhone: ['', [Validators.required, Validators.pattern('^\\d{3}-\\d{3}-\\d{4}$')]],
      email: ['', [Validators.required, Validators.email]],
      // Add other fields accordingly...
    });
  }

  ngOnInit(): void {
    // Any initialization logic if necessary
  }

  onSubmit(): void {
    if (this.clientProfileForm.valid) {
      this.isSubmitting = true;
      const clientProfile: ClientProfile = this.clientProfileForm.value;
      // Call the service to save the client profile
      this.clientProfileService.createProfile(clientProfile).subscribe({
        next: (response) => {
          console.log('Profile created successfully:', response);
          this.isSubmitting = false;
          // Optionally reset the form or navigate to another route
        },
        error: (error) => {

        }
