```python
class ClientProfile:
    def __init__(self, client_id, name, email, phone):
        self.client_id = client_id
        self.name = name
        self.email = email
        self.phone = phone

    def __repr__(self):
        return f"ClientProfile(client_id={self.client_id}, name='{self.name}', email='{self.email}', phone='{self.phone}')"


class ClientProfileManager:
    def __init__(self):
        self.profiles = {}
        self.next_id = 1

    def create_profile(self, name, email, phone):
        client_id = self.next_id
        self.profiles[client_id] = ClientProfile(client_id, name, email, phone)
        self.next_id += 1
        return client_id

    def update_profile(self, client_id, name=None, email=None, phone=None):
        if client_id not in self.profiles:
            raise ValueError("Client ID not found.")
        
        profile = self.profiles[client_id]
        if name:
            profile.name = name
        if email:
            profile.email = email
        if phone:
            profile.phone = phone

    def delete_profile(self, client_id):
        if client_id not in self.profiles:
            raise ValueError("Client ID not found.")
        del self.profiles[client_id]

    def retrieve_profile(self, client_id):
        if client_id not in self.profiles:
            raise ValueError("Client ID not found.")
        return self.profiles[client_id]

    def list_all_profiles(self):
        return list(self.profiles.values())


# Example usage
if __name__ == "__main__":
    manager = ClientProfileManager()
    client_id = manager.create_profile("John Doe", "john@example.com", "123-456-7890")
    print(manager.retrieve_profile(client_id))  # Retrieve client profile
    manager.update_profile(client_id, phone="098-765-4321")  # Update phone number
    print(manager.list_all_profiles())  # List all profiles
    manager.delete_profile(client_id)  # Delete profile
```

- **ClientProfileManager Class**: Contains methods for managing client profiles, including creating, updating, deleting, and retrieving profiles.
- **Usage Example**: Demonstrates how to create, update, retrieve, list, and delete client profiles.
