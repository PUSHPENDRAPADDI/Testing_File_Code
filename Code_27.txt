### FastAPI Implementation

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field
from typing import List, Optional

app = FastAPI()

# Sample Data Storage (In-memory for demonstration purposes)
patients = {}
appointments = {}
medical_records = {}
prescriptions = {}
invoices = {}

# Patient Model
class Patient(BaseModel):
    id: int
    name: str
    contact_details: str
    medical_history: str

# Appointment Model
class Appointment(BaseModel):
    id: int
    patient_id: int
    doctor_id: int
    date_time: str
    status: str

# Medical Record Model
class MedicalRecord(BaseModel):
    patient_id: int
    record: str

# Prescription Model
class Prescription(BaseModel):
    id: int
    patient_id: int
    details: str

# Invoice Model
class Invoice(BaseModel):
    id: int
    patient_id: int
    amount: float
    status: str

# Patient Management API
@app.post("/patients", response_model=Patient)
def register_patient(patient: Patient):
    if patient.id in patients:
        raise HTTPException(status_code=400, detail="Patient already registered.")
    patients[patient.id] = patient
    return patient

@app.get("/patients/{id}", response_model=Patient)
def retrieve_patient(id: int):
    if id not in patients:
        raise HTTPException(status_code=404, detail="Patient not found.")
    return patients[id]

@app.put("/patients/{id}", response_model=Patient)
def update_patient(id: int, patient: Patient):
    if id not in patients:
        raise HTTPException(status_code=404, detail="Patient not found.")
    patients[id] = patient
    return patient

@app.delete("/patients/{id}", response_model=dict)
def deactivate_patient(id: int):
    if id not in patients:
        raise HTTPException(status_code=404, detail="Patient not found.")
    del patients[id]
    return {"message": "Patient deactivated successfully."}

# Appointment Scheduling API
@app.post("/appointments", response_model=Appointment)
def schedule_appointment(appointment: Appointment):
    if appointment.id in appointments:
        raise HTTPException(status_code=400, detail="Appointment already scheduled.")
    appointments[appointment.id] = appointment
    return appointment

@app.put("/appointments/{id}", response_model=Appointment)
def modify_appointment(id: int, appointment: Appointment):
    if id not in appointments:
        raise HTTPException(status_code=404, detail="Appointment not found.")
    appointments[id] = appointment
    return appointment

@app.get("/doctors/{id}/availability")
def retrieve_doctor_availability(id: int):
    # Placeholder logic for doctor availability
    return {"doctor_id": id, "available": True}

# EMR API
@app.post("/patients/{id}/records", response_model=MedicalRecord)
def add_medical_record(id: int, record: MedicalRecord):
    if id not in patients:
        raise HTTPException(status_code=404, detail="Patient not found.")
    medical_records.setdefault(id, []).append(record)
    return record

@app.get("/patients/{id}/records", response_model=List[MedicalRecord])
def retrieve_medical_records(id: int):
    if id not in patients:
        raise HTTPException(status_code=404, detail="Patient not found.")
    return medical_records.get(id, [])

# Prescription Management API
@app.post("/prescriptions", response_model=Prescription)
def generate_prescription(prescription: Prescription):
    prescriptions[prescription.id] = prescription
    return prescription

@app.get("/prescriptions/{id}", response_model=Prescription)
def retrieve_prescription(id: int):
    if id not in prescriptions:
        raise HTTPException(status_code=404, detail="Prescription not found.")
    return prescriptions[id]

# Billing & Insurance API
@app.post("/billing", response_model=Invoice)
def generate_invoice(invoice: Invoice):
    invoices[invoice.id] = invoice
    return invoice

@app.get("/billing/{id}", response_model=Invoice)
def retrieve_invoice(id: int):
    if id not in invoices:
        raise HTTPException(status_code=404, detail="Invoice not found.")
    return invoices[id]

@app.post("/insurance/claims")
def process_insurance_claim(claim: dict):

    return {"message": "Insurance claim processed successfully."}
```

### How to Run the Application
1. Install FastAPI and Uvicorn:
   ```bash
   pip install fastapi uvicorn
   ```

2. Save the code above in a file, e.g., `app.py`.

3. Run the FastAPI application using Uvicorn:
   ```bash
   uvicorn app:app --reload
   ```

4. Open your browser and navigate to `http://127.0.0.1:8000/docs` to see the automatically generated API documentation and test the endpoints.
