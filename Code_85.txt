import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/client-profiles")
public class ClientProfileController {
    @Autowired
    private ClientProfileService service;

    @PostMapping
    public ResponseEntity<ClientProfile> createClientProfile(@RequestBody ClientProfile clientProfile) {
        return ResponseEntity.ok(service.createClientProfile(clientProfile));
    }

    @GetMapping("/{id}")
    public ResponseEntity<ClientProfile> getClientProfile(@PathVariable Long id) {
        return service.getClientProfile(id)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }

    @PutMapping("/{id}")
    public ResponseEntity<ClientProfile> updateClientProfile(@PathVariable Long id, @RequestBody ClientProfile clientProfile) {
        return ResponseEntity.ok(service.updateClientProfile(id, clientProfile));
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> deleteClientProfile(@PathVariable Long id) {
        service.deleteClientProfile(id);
        return ResponseEntity.noContent().build();
    }

    @GetMapping
    public ResponseEntity<List<ClientProfile>> getAllClientProfiles() {
        return ResponseEntity.ok(service.getAllClientProfiles());
