### FastAPI Code Implementation

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional
from uuid import uuid4

app = FastAPI()

# In-memory data storage
patients_db = {}
appointments_db = {}
records_db = {}
prescriptions_db = {}
billing_db = {}

# Models
class Patient(BaseModel):
    name: str
    contact: str
    medical_history: Optional[str] = None

class Appointment(BaseModel):
    patient_id: str
    doctor_id: str
    date: str
    time: str

class MedicalRecord(BaseModel):
    record_id: str
    details: str

class Prescription(BaseModel):
    patient_id: str
    doctor_id: str
    prescription_details: str

class Invoice(BaseModel):
    patient_id: str
    amount: float
    status: str

# Patient Management API
@app.post("/patients", response_model=str)
def register_patient(patient: Patient):
    patient_id = str(uuid4())
    patients_db[patient_id] = patient
    return patient_id

@app.get("/patients/{id}", response_model=Patient)
def get_patient(id: str):
    if id not in patients_db:
        raise HTTPException(status_code=404, detail="Patient not found")
    return patients_db[id]

@app.put("/patients/{id}", response_model=Patient)
def update_patient(id: str, patient: Patient):
    if id not in patients_db:
        raise HTTPException(status_code=404, detail="Patient not found")
    patients_db[id] = patient
    return patient

@app.delete("/patients/{id}")
def deactivate_patient(id: str):
    if id not in patients_db:
        raise HTTPException(status_code=404, detail="Patient not found")
    del patients_db[id]
    return {"message": "Patient profile deactivated"}

# Appointment Scheduling API
@app.post("/appointments", response_model=str)
def schedule_appointment(appointment: Appointment):
    appointment_id = str(uuid4())
    appointments_db[appointment_id] = appointment
    return appointment_id

@app.put("/appointments/{id}")
def modify_appointment(id: str, appointment: Appointment):
    if id not in appointments_db:
        raise HTTPException(status_code=404, detail="Appointment not found")
    appointments_db[id] = appointment
    return {"message": "Appointment updated"}

@app.get("/doctors/{id}/availability")
def get_doctor_availability(id: str):
    # Implement logic for availability checking here
    return {"available": True}  # Placeholder

# EMR API
@app.get("/patients/{id}/records", response_model=List[MedicalRecord])
def get_medical_records(id: str):
    return records_db.get(id, [])

@app.post("/patients/{id}/records")
def add_update_medical_record(id: str, record: MedicalRecord):
    if id not in records_db:
        records_db[id] = []
    records_db[id].append(record)
    return {"message": "Record added/updated"}

# Prescription Management API
@app.post("/prescriptions")
def generate_prescription(prescription: Prescription):
    prescription_id = str(uuid4())
    prescriptions_db[prescription_id] = prescription
    return prescription_id

@app.get("/prescriptions/{id}")
def retrieve_prescription(id: str):
    if id not in prescriptions_db:
        raise HTTPException(status_code=404, detail="Prescription not found")
    return prescriptions_db[id]

# Billing & Insurance API
@app.post("/billing")
def generate_invoice(invoice: Invoice):
    invoice_id = str(uuid4())
    billing_db[invoice_id] = invoice
    return invoice_id

@app.get("/billing/{id}")
def retrieve_invoice(id: str):
    if id not in billing_db:
        raise HTTPException(status_code=404, detail="Invoice not found")
    return billing_db[id]
```

### Running the Application

1. Save the code to a file named `app.py`.
2. Install FastAPI and an ASGI server (like Uvicorn) if you haven't already:

   ```bash
   pip install fastapi uvicorn
   ```

3. Run the application using Uvicorn:

   ```bash
   uvicorn app:app --reload
   ```

4. You can access the API documentation at `http://127.0.0.1:8000/docs` to test the endpoints.
