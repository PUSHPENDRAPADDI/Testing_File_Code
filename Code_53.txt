// Controllers/ClientProfileController.cs
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace YourNamespace.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly IClientProfileManagement _clientProfileManagement;

        public ClientProfileController(IClientProfileManagement clientProfileManagement)
        {
            _clientProfileManagement = clientProfileManagement;
        }

        [HttpPost("create")]
        public IActionResult CreateClientProfile(ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile cannot be null");
            }

            _clientProfileManagement.CreateClientProfile(clientProfile);
            return Ok("Client profile created successfully");
        }

        [HttpGet("{id}")]
        public IActionResult GetClientProfile(int id)
        {
            var clientProfile = _clientProfileManagement.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }
            return Ok(clientProfile);
        }

        [HttpPut("update")]
        public IActionResult UpdateClientProfile(ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile cannot be null");
            }

            _clientProfileManagement.Update(clientProfile);
            return Ok("Client profile updated successfully");
        }

        [HttpDelete("delete/{id}")]
        public IActionResult DeleteClientProfile(int id)
        {
            var clientProfile = _clientProfileManagement.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }

            _clientProfileManagement.Delete(clientProfile);
            return Ok("Client profile deleted successfully");
        }
    }
}
