```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;
using System.Collections.Generic;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly ICustomService<ClientProfile> _customService;
        private readonly ApplicationDbContext _applicationDbContext;

        public ClientProfileController(ICustomService<ClientProfile> customService, ApplicationDbContext applicationDbContext)
        {
            _customService = customService;
            _applicationDbContext = applicationDbContext;
        }

        [HttpPost(nameof(CreateClientProfile))]
        public IActionResult CreateClientProfile(ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Invalid Client Profile Data");
            }

            _customService.Insert(clientProfile);
            return Ok("Client Profile Created Successfully");
        }

        [HttpGet(nameof(GetClientProfileById))]
        public IActionResult GetClientProfileById(int id)
        {
            var profile = _customService.Get(id);
            if (profile == null)
            {
                return NotFound();
            }
            return Ok(profile);
        }

        [HttpPut(nameof(UpdateClientProfile))]
        public IActionResult UpdateClientProfile(int id, ClientProfile clientProfile)
        {
            if (clientProfile == null || clientProfile.Id != id)
            {
                return BadRequest("Invalid Client Profile Data");
            }

            var existingProfile = _customService.Get(id);
            if (existingProfile == null)
            {
                return NotFound();
            }

            _customService.Update(clientProfile);
            return Ok("Client Profile Updated Successfully");
        }

        [HttpDelete(nameof(DeleteClientProfile))]
        public IActionResult DeleteClientProfile(int id)
        {
            var existingProfile = _customService.Get(id);
            if (existingProfile == null)
            {
                return NotFound();
            }

            _customService.Delete(id);
            return Ok("Client Profile Deleted Successfully");
        }

        [HttpGet(nameof(GetAllClientProfiles))]
        public IActionResult GetAllClientProfiles()
        {
            var profiles = _customService.GetAll();
            return Ok(profiles);
        }
    }
}
```
