### Step 2: Create the Server

Create a file named `server.js` and add the following code:

```javascript
const express = require('express');
const bodyParser = require('body-parser');

const app = express();
const port = 3000;

// Middleware to parse JSON requests
app.use(bodyParser.json());

// In-memory database for client profiles
let clientProfiles = [];
let currentId = 1;

// Create client profile
app.post('/profiles', (req, res) => {
    const { name, email, phone } = req.body;
    const newProfile = { id: currentId++, name, email, phone };
    clientProfiles.push(newProfile);
    res.status(201).json(newProfile);
});

// Retrieve all client profiles
app.get('/profiles', (req, res) => {
    res.json(clientProfiles);
});

// Retrieve a client profile by ID
app.get('/profiles/:id', (req, res) => {
    const profile = clientProfiles.find(p => p.id === parseInt(req.params.id));
    if (profile) {
        res.json(profile);
    } else {
        res.status(404).send('Profile not found');
    }
});

// Update a client profile by ID
app.put('/profiles/:id', (req, res) => {
    const profile = clientProfiles.find(p => p.id === parseInt(req.params.id));
    if (profile) {
        const { name, email, phone } = req.body;
        profile.name = name !== undefined ? name : profile.name;
        profile.email = email !== undefined ? email : profile.email;
        profile.phone = phone !== undefined ? phone : profile.phone;
        res.json(profile);
    } else {
        res.status(404).send('Profile not found');
    }
});

// Delete a client profile by ID
app.delete('/profiles/:id', (req, res) => {
    const index = clientProfiles.findIndex(p => p.id === parseInt(req.params.id));
    if (index !== -1) {
        clientProfiles.splice(index, 1);
        res.status(204).send();
    } else {
        res.status(404).send('Profile not found');
    }
});

// Start the server
app.listen(port, () => {
    console.log(`Server running at http://localhost:${port}`);
});
```
