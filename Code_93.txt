using RepositoryLayer.IRepository;
using ServiceLayer.ICustomServices;

namespace ServiceLayer.CustomServices
{
    public class ClientProfileService : ICustomService<ClientProfile>
    {
        private readonly IRepository<ClientProfile> _clientProfileRepository;

        public ClientProfileService(IRepository<ClientProfile> clientProfileRepository)
        {
            _clientProfileRepository = clientProfileRepository;
        }

        public void Insert(ClientProfile entity)
        {
            _clientProfileRepository.Insert(entity);
            _clientProfileRepository.SaveChanges();
        }

        public ClientProfile Get(int id)
        {
            return _clientProfileRepository.Get(id);
        }

        public void Update(ClientProfile entity)
        {
            _clientProfileRepository.Update(entity);
            _clientProfileRepository.SaveChanges();
        }

        public void Delete(int id)
        {
            var clientProfile = _clientProfileRepository.Get(id);
            if (clientProfile != null)
            {
                _clientProfileRepository.Delete(clientProfile);
                _clientProfileRepository.SaveChanges();
            }
        }

    }
}
