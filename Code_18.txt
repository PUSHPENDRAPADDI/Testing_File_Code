using DomainLayer.Models;
using RepositoryLayer.IRepository;
using ServiceLayer.ICustomServices;

namespace ServiceLayer.CustomServices

    {
        private readonly IRepository<ClientProfile> _clientProfileRepository;

        public ClientProfileManagement(IRepository<ClientProfile> clientProfileRepository)
        {
            _clientProfileRepository = clientProfileRepository ?? throw new ArgumentNullException(nameof(clientProfileRepository));
        }

        public void Insert(ClientProfile entity)
        {
            if (entity == null)
            {
                throw new ArgumentNullException(nameof(entity), "ClientProfile cannot be null.");
            }

            ValidateClientProfile(entity);

            _clientProfileRepository.Insert(entity);
            _clientProfileRepository.SaveChanges();
        }

        private void ValidateClientProfile(ClientProfile clientProfile)
        {
            // Example validation logic, may vary based on the actual properties of ClientProfile
            if (string.IsNullOrWhiteSpace(clientProfile.Name))
            {
                throw new ArgumentException("Client name is required.", nameof(clientProfile.Name));
            }

            if (string.IsNullOrWhiteSpace(clientProfile.Email) || !IsValidEmail(clientProfile.Email))
            {
                throw new ArgumentException("A valid email is required.", nameof(clientProfile.Email));
            }

            // Additional validation rules can be added here
        }

        private bool IsValidEmail(string email)
        {
            // Simple email validation. Consider using a more robust regex for production code.
            return email.Contains("@");
        }
    }
}
