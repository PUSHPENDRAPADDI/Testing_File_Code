**Repository Implementation:**

```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using RepositoryLayer.IRepository;
using Microsoft.EntityFrameworkCore;
using System.Collections.Generic;
using System.Linq;

namespace RepositoryLayer.Repository
{
    public class ClientProfileRepository : IClientProfileRepository
    {
        private readonly ApplicationDbContext _context;
        private DbSet<ClientProfile> _entities;

        public ClientProfileRepository(ApplicationDbContext context)
        {
            _context = context;
            _entities = _context.Set<ClientProfile>();
        }

        public ClientProfile Get(int id)
        {
            return _entities.SingleOrDefault(c => c.Id == id);
        }

        public IEnumerable<ClientProfile> GetAll()
        {
            return _entities.AsEnumerable();
        }

        public void Insert(ClientProfile clientProfile)
        {
            _entities.Add(clientProfile);
            _context.SaveChanges();
        }

        public void Update(ClientProfile clientProfile)
        {
            _entities.Update(clientProfile);
            _context.SaveChanges();
        }

        public void Delete(ClientProfile clientProfile)
        {
            _entities.Remove(clientProfile);
            _context.SaveChanges();
        }

        public void SaveChanges()
        {
            _context.SaveChanges();
        }
    }
}
```

### 4. Controller Layer

**Controller Implementation:**

```csharp
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly IClientProfileService _service;

        public ClientProfileController(IClientProfileService service)
        {
            _service = service;
        }

        [HttpGet("{id}")]
        public IActionResult GetClientProfile(int id)
        {
            var clientProfile = _service.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }
            return Ok(clientProfile);
        }

        [HttpGet]
        public IActionResult GetAllClientProfiles()
        {
            var clientProfiles = _service.GetAll();
            return Ok(clientProfiles);
        }

        [HttpPost]
        public IActionResult CreateClientProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile data is null.");
            }
            _service.Create(clientProfile);
            return CreatedAtAction(nameof(GetClientProfile), new { id = clientProfile.Id }, clientProfile);
        }

        [HttpPut("{id}")]
        public IActionResult UpdateClientProfile(int id, [FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null || clientProfile.Id != id)
            {
                return BadRequest("Client profile data is invalid.");
            }
            _service.Update(clientProfile);
            return NoContent();
        }

        [HttpDelete("{id}")]
        public IActionResult DeleteClientProfile(int id)
        {
            _service.Delete(id);
            return NoContent();
        }
    }
}
```
