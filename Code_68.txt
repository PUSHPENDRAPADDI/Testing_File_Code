```csharp
using HealthcareService.Models;
using System.Collections.Generic;
using System.Threading.Tasks;

namespace HealthcareService.Repositories
```

### Step 3: Implement the Repository

Now, let's implement this repository. Here, I'll use Entity Framework Core as an example for database operations.

```csharp
// Repositories/ClientProfileRepository.cs
using HealthcareService.Models;
using Microsoft.EntityFrameworkCore;
using System.Collections.Generic;
using System.Threading.Tasks;

namespace HealthcareService.Repositories
{
    public class ClientProfileRepository : IClientProfileRepository
    {
        private readonly ApplicationDbContext _context;

        public ClientProfileRepository(ApplicationDbContext context)
        {
            _context = context;
        }

        public async Task<List<ClientProfile>> GetAllProfilesAsync()
        {
            return await _context.ClientProfiles.ToListAsync();
        }

        public async Task<ClientProfile> GetProfileByIdAsync(int id)
        {
            return await _context.ClientProfiles.FindAsync(id);
        }

        public async Task AddProfileAsync(ClientProfile profile)
        {
            profile.CreatedDate = DateTime.UtcNow;
            await _context.ClientProfiles.AddAsync(profile);
            await _context.SaveChangesAsync();
        }

        public async Task UpdateProfileAsync(ClientProfile profile)
        {
            profile.UpdatedDate = DateTime.UtcNow;
            _context.ClientProfiles.Update(profile);
            await _context.SaveChangesAsync();
        }

        public async Task DeleteProfileAsync(int id)
        {
            var profile = await GetProfileByIdAsync(id);
            if (profile != null)
            {
                _context.ClientProfiles.Remove(profile);
                await _context.SaveChangesAsync();
            }
        }

        public async Task<int> GetCompletionPercentageAsync(int id)
        {
            var profile = await GetProfileByIdAsync(id);
            if (profile == null) return 0;

            int totalFields = 10; // Adjust based on the total number of fields
            int filledFields = 0;

            if (!string.IsNullOrWhiteSpace(profile.FirstName) ) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.LastName)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.Address)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.ContactNumber)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.InsuranceDetails)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.ClinicPreference)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.MedicalHistory)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.OngoingTreatment)) filledFields++;
            if (!string.IsNullOrWhiteSpace(profile.Status)) filledFields++;

            return (filledFields * 100) / totalFields;
        }
    }
}
```

### Step 6: Create Controller

Finally, create a controller to handle HTTP requests related to client profiles.

```csharp
// Controllers/ClientProfileController.cs
using HealthcareService.Models;
using HealthcareService.Repositories;
using Microsoft.AspNetCore.Mvc;
using System.Threading.Tasks;

namespace HealthcareService.Controllers
{
    public class ClientProfileController : Controller
    {
        private readonly IClientProfileRepository _repository;

        public ClientProfileController(IClientProfileRepository repository)
        {
            _repository = repository;
        }

        public async Task<IActionResult> Index()
        {
            var profiles = await _repository.GetAllProfilesAsync();
            return View(profiles);
        }

        public async Task<IActionResult> Details(int id)
        {
            var profile = await _repository.GetProfileByIdAsync(id);
            if (profile == null) return NotFound();
            return View(profile);
        }

        [HttpPost]
        public async Task<IActionResult> Create(ClientProfile profile)
        {
            if (ModelState.IsValid)
            {
                await _repository.AddProfileAsync(profile);
                return RedirectToAction(nameof(Index));
            }
            return View(profile);
        }

        // Implement other actions for Edit and Delete
    }
}
```
