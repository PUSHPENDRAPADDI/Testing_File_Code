using DomainLayer.Data;

using System.Collections.Generic;

namespace RepositoryLayer.Repository
{
    public class ClientProfileRepository : IRepository<ClientProfile>
    {
        private readonly ApplicationDbContext _applicationDbContext;
        private DbSet<ClientProfile> entities;

        public ClientProfileRepository(ApplicationDbContext applicationDbContext)
        {
            _applicationDbContext = applicationDbContext;
            entities = _applicationDbContext.Set<ClientProfile>();
        }

        public void Insert(ClientProfile entity)
        {
            entities.Add(entity);
        }

        public ClientProfile Get(int id)
        {
            return entities.SingleOrDefault(e => e.Id == id);
        }

        public IEnumerable<ClientProfile> GetAll()
        {
            return entities.ToList();
        }

        public void Update(ClientProfile entity)
        {
            // Attach the entity to the context and mark it as modified
            entities.Attach(entity);
            _applicationDbContext.Entry(entity).State = EntityState.Modified;
        }

        public void Delete(int id)
        {
            var entity = Get(id);
            if(entity != null)
            {
                entities.Remove(entity);
            }
        }

        public void SaveChanges()
        {
            _applicationDbContext.SaveChanges();
        }
    }
}
