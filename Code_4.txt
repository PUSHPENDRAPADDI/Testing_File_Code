```csharp
using System.Collections.Generic;
using System.Threading.Tasks;
using Microsoft.EntityFrameworkCore;

public class ClientProfileRepository : IClientProfileRepository
{
    private readonly ApplicationDbContext _context;

    public ClientProfileRepository(ApplicationDbContext context)
    {
        _context = context;
    }

    public async Task<ClientProfile> GetClientProfileByIdAsync(int id)
    {
        return await _context.ClientProfiles.FindAsync(id);
    }

    public async Task<IEnumerable<ClientProfile>> GetAllClientProfilesAsync()
    {
        return await _context.ClientProfiles.ToListAsync();
    }

    public async Task CreateClientProfileAsync(ClientProfile clientProfile)
    {
        await _context.ClientProfiles.AddAsync(clientProfile);
        await _context.SaveChangesAsync();
    }

    public async Task UpdateClientProfileAsync(ClientProfile clientProfile)
    {
        _context.ClientProfiles.Update(clientProfile);
        await _context.SaveChangesAsync();
    }

    public async Task DeleteClientProfileAsync(int id)
    {
        var clientProfile = await GetClientProfileByIdAsync(id);
        if (clientProfile != null)
        {
            _context.ClientProfiles.Remove(clientProfile);
            await _context.SaveChangesAsync();
        }
    }
}
```
