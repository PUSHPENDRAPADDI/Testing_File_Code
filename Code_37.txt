### 3. Repository Implementation

**ClientProfileRepository.cs** (located in `RepositoryLayer.Repository`)

```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using RepositoryLayer.IRepository;
using Microsoft.EntityFrameworkCore;
using System.Collections.Generic;

namespace RepositoryLayer.Repository
{
    public class ClientProfileRepository : IClientProfileRepository
    {
        private readonly ApplicationDbContext _context;
        private DbSet<ClientProfile> _entities;

        public ClientProfileRepository(ApplicationDbContext context)
        {
            _context = context;
            _entities = _context.Set<ClientProfile>();
        }

        public void Insert(ClientProfile entity)
        {
            _entities.Add(entity);
            SaveChanges();
        }

        public void Update(ClientProfile entity)
        {
            _entities.Update(entity);
            SaveChanges();
        }

        public void Delete(ClientProfile entity)
        {
            _entities.Remove(entity);
            SaveChanges();
        }

        public ClientProfile Get(int id)
        {
            return _entities.SingleOrDefault(e => e.Id == id);
        }

        public IEnumerable<ClientProfile> GetAll()
        {
            return _entities.AsEnumerable();
        }

        public void SaveChanges()
        {
            _context.SaveChanges();
        }
    }
}
```

### 6. Controller

**ClientProfileController.cs** (located in `Controllers`)

```csharp
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly IClientProfileService _clientProfileService;

        public ClientProfileController(IClientProfileService clientProfileService)
        {
            _clientProfileService = clientProfileService;
        }

        [HttpPost("CreateClientProfile")]
        public IActionResult CreateClientProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Invalid client profile data.");
            }

            _clientProfileService.Insert(clientProfile);
            return Ok("Client profile created successfully.");
        }

        [HttpPut("UpdateClientProfile/{id}")]
        public IActionResult UpdateClientProfile(int id, [FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null || clientProfile.Id != id)
            {
                return BadRequest("Invalid client profile data.");
            }

            var existingProfile = _clientProfileService.Get(id);
            if (existingProfile == null)
            {
                return NotFound("Client profile not found.");
            }

            _clientProfileService.Update(clientProfile);
            return Ok("Client profile updated successfully.");
        }

        [HttpGet("GetClientProfile/{id}")]
        public IActionResult GetClientProfile(int id)
        {
            var profile = _clientProfileService.Get(id);
            if (profile == null)
            {
                return NotFound();
            }
            return Ok(profile);
        }

        [HttpDelete("DeleteClientProfile/{id}")]
        public IActionResult DeleteClientProfile(int id)
        {
            var profile = _clientProfileService.Get(id);
            if (profile == null)
            {
                return NotFound();
            }

            _clientProfileService.Delete(profile);
            return Ok("Client profile deleted successfully.");
        }
    }
}
```
