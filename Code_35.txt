```typescript
import { AfterViewInit, ChangeDetectorRef, Component, OnDestroy, OnInit } from '@angular/core';
import { MatDialog } from '@angular/material/dialog';
import { MatTableDataSource } from '@angular/material/table';
import { UserDetailService } from '@core/services/user-detail.service';
import { GenericListDataModel, columnDefination, commonMSG, DashboardCardData } from '@shared/models/common.model';
import { AlertMessageService } from '@shared/services/alert-message/alert-message.service';
import { ApiList } from '@shared/services/api-list';
import { ApiService } from '@shared/services/api.service';
import { Subscription, of, throwError } from 'rxjs';
import { catchError } from 'rxjs/operators';
import moment from 'moment';

@Component({
  selector: 'app-client-profile',
  templateUrl: './client-profile.component.html',
  styleUrls: ['./client-profile.component.scss']
})
export class ClientProfileComponent implements OnInit, AfterViewInit, OnDestroy {
  gridDataSource: MatTableDataSource<any> = new MatTableDataSource();
  model: GenericListDataModel = {} as GenericListDataModel;
  isLoaded: boolean = false;
  selectedTabIndex: number = 0;
  dashboardData: DashboardCardData | undefined | null;
  notificationSubscription: Subscription | undefined;

  constructor(
    private cd: ChangeDetectorRef,
    private alertMessageService: AlertMessageService,
    public api: ApiService,
    private dialog: MatDialog,
    private userDetailService: UserDetailService
  ) {}

  ngOnInit() {
    this.notificationSubscription = this.dataService.notificationAlertsData$.subscribe(result => {
      if (result) {
        this.dashboardData = result;
        this.setTabIndex(result.selectedTab);
        this.loadData();
      }
    });
  }

  ngAfterViewInit() {
    this.loadData();
  }

  loadData() {
    switch (this.selectedTabIndex) {
      case 0:
        this.loadInquiryData();
        break;
      case 1:
        this.loadClientProfileData();
        break;
      case 2:
        this.loadMedicalNecessityData();
        break;
      default:
        break;
    }
  }

  loadInquiryData() {
    this.isLoaded = false;
    this.api.getData(ApiList.inquiry.getInquiries).pipe(
      catchError(err => {
        this.alertMessageService.ErrorSnackBar('Failed to load inquiries');
        return throwError(() => err);
      })
    ).subscribe(res => {
      this.gridDataSource.data = res.inquiries || [];
      this.isLoaded = true;
      this.cd.detectChanges();
    });
  }

  loadClientProfileData() {
    this.isLoaded = false;
    this.api.getData(ApiList.clientProfile.getClientProfiles).pipe(
      catchError(err => {
        this.alertMessageService.ErrorSnackBar('Failed to load client profiles');
        return throwError(() => err);
      })
    ).subscribe(res => {
      this.gridDataSource.data = res.clientProfiles || [];
      this.isLoaded = true;
      this.cd.detectChanges();
    });
  }

  loadMedicalNecessityData() {
    this.isLoaded = false;
    this.api.getData(ApiList.medicalNecessity.getMedicalNecessity).pipe(
      catchError(err => {
        this.alertMessageService.ErrorSnackBar('Failed to load medical necessities');
        return throwError(() => err);
      })
    ).subscribe(res => {
      this.gridDataSource.data = res.medicalNecessity || [];
      this.isLoaded = true;
      this.cd.detectChanges();
    });
  }

  setTabIndex(index: number) {
    this.selectedTabIndex = index;
    this.loadData();
  }

  ngOnDestroy(): void {
    this.notificationSubscription?.unsubscribe();
  }
}
```
