### Express.js Healthcare API Implementation

1. **Setup and Installation**

First, ensure you have Node.js and npm installed on your machine. Then, create a new project folder and initialize a new Node.js application.

```bash
mkdir healthcare-api
cd healthcare-api
npm init -y
npm install express body-parser cors
```

2. **Create Project Structure**

Create the following folder structure:

```
healthcare-api/
├── index.js
├── routes/
│   ├── patient.js
│   ├── appointment.js
│   ├── emr.js
```

3. **Implementation**

Now we will implement the basic routes in each file.

### `index.js`

```javascript
const express = require('express');
const bodyParser = require('body-parser');
const cors = require('cors');

const patientRoutes = require('./routes/patient');
const appointmentRoutes = require('./routes/appointment');
const emrRoutes = require('./routes/emr');
const prescriptionRoutes = require('./routes/prescription');
const billingRoutes = require('./routes/billing');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(cors());
app.use(bodyParser.json());

// Use routes
app.use('/patients', patientRoutes);
app.use('/appointments', appointmentRoutes);
app.use('/emr', emrRoutes);
app.use('/prescriptions', prescriptionRoutes);
app.use('/billing', billingRoutes);

app.listen(PORT, () => {
    console.log(`Server is running on port ${PORT}`);
});
```

### `routes/patient.js`

```javascript
const express = require('express');
const router = express.Router();

// Mock data for demonstration
let patients = [];

// Register a new patient
router.post('/', (req, res) => {
    const newPatient = { id: patients.length + 1, ...req.body };
    patients.push(newPatient);
    res.status(201).json(newPatient);
});

// Retrieve patient details
router.get('/:id', (req, res) => {
    const patient = patients.find(p => p.id === parseInt(req.params.id));
    if (!patient) return res.status(404).send('Patient not found');
    res.json(patient);
});

// Update patient details
router.put('/:id', (req, res) => {
    const patient = patients.find(p => p.id === parseInt(req.params.id));
    if (!patient) return res.status(404).send('Patient not found');
    Object.assign(patient, req.body);
    res.json(patient);
});

// Deactivate patient profile
router.delete('/:id', (req, res) => {
    const patientIndex = patients.findIndex(p => p.id === parseInt(req.params.id));
    if (patientIndex === -1) return res.status(404).send('Patient not found');
    patients.splice(patientIndex, 1);
    res.status(204).send();
});

module.exports = router;
```

### `routes/emr.js`

```javascript
const express = require('express');
const router = express.Router();

// Mock data for demonstration
let medicalRecords = [];

router.get('/patients/:id/records', (req, res) => {
    const records = medicalRecords.filter(record => record.patientId === parseInt(req.params.id));
    res.json(records);
});

// Add/update medical records

    const newRecord = { id: medicalRecords.length + 1, patientId: parseInt(req.params.id), ...req.body };
    medicalRecords.push(newRecord);
    res.status(201).json(newRecord);
});

module.exports = router;
```
