### 6. Controller

Finally, we will create the controller for `ClientProfile`.

```csharp
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly IClientProfileService _clientProfileService;

        public ClientProfileController(IClientProfileService clientProfileService)
        {
            _clientProfileService = clientProfileService;
        }

        [HttpPost(nameof(CreateClientProfile))]
        public IActionResult CreateClientProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile cannot be null.");
            }

            _clientProfileService.Insert(clientProfile);
            return Ok("Client profile created successfully.");
        }

        [HttpGet(nameof(GetClientProfileById))]
        public IActionResult GetClientProfileById(int id)
        {
            var clientProfile = _clientProfileService.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }

            return Ok(clientProfile);
        }

        [HttpGet(nameof(GetAllClientProfiles))]
        public IActionResult GetAllClientProfiles()
        {
            var clientProfiles = _clientProfileService.GetAll();
            return Ok(clientProfiles);
        }

        [HttpPut(nameof(UpdateClientProfile))]
        public IActionResult UpdateClientProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile cannot be null.");
            }

            _clientProfileService.Update(clientProfile);
            return Ok("Client profile updated successfully.");
        }

        [HttpDelete(nameof(DeleteClientProfile))]
        public IActionResult DeleteClientProfile(int id)
        {
            var clientProfile = _clientProfileService.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }

            _clientProfileService.Delete(clientProfile);
            return Ok("Client profile deleted successfully.");
        }
    }
}
```
