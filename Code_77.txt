### 3. Repository Layer

This layer will handle the data access logic.

```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using RepositoryLayer.IRepository;
using System.Collections.Generic;
using System.Linq;

namespace RepositoryLayer.Repository
{
    public class ClientProfileRepository : IRepository<ClientProfile>
    {
        private readonly ApplicationDbContext _context;

        public ClientProfileRepository(ApplicationDbContext context)
        {
            _context = context;
        }

        public void Insert(ClientProfile entity)
        {
            _context.ClientProfiles.Add(entity);
            _context.SaveChanges();
        }

        public ClientProfile Get(int id)
        {
            return _context.ClientProfiles.FirstOrDefault(c => c.Id == id);
        }

        public IEnumerable<ClientProfile> GetAll()
        {
            return _context.ClientProfiles.ToList();
        }

        public void Update(ClientProfile entity)
        {
            _context.ClientProfiles.Update(entity);
            _context.SaveChanges();
        }

        public void Delete(ClientProfile entity)
        {
            _context.ClientProfiles.Remove(entity);
            _context.SaveChanges();
        }
    }
}
```

### 5. Controller

This controller will handle HTTP requests related to client profiles.

```csharp
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly ICustomService<ClientProfile> _service;

        public ClientProfileController(ICustomService<ClientProfile> service)
        {
            _service = service;
        }

        [HttpPost("create")]
        public IActionResult CreateProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile is null.");
            }

            _service.Insert(clientProfile);
            return CreatedAtAction("GetProfile", new { id = clientProfile.Id }, clientProfile);
        }

        [HttpGet("{id}")]
        public IActionResult GetProfile(int id)
        {
            var profile = _service.Get(id);
            if (profile == null)
            {
                return NotFound();
            }
            return Ok(profile);
        }

        [HttpGet]
        public IActionResult GetAllProfiles()
        {
            return Ok(_service.GetAll());
        }

        [HttpPut("{id}")]
        public IActionResult UpdateProfile(int id, [FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null || clientProfile.Id != id)
            {
                return BadRequest();
            }

            _service.Update(clientProfile);
            return NoContent();
        }

        [HttpDelete("{id}")]
        public IActionResult DeleteProfile(int id)
        {
            var profile = _service.Get(id);
            if (profile == null)
            {
                return NotFound();
            }

            _service.Delete(profile);
            return NoContent();
        }
    }
}
```
