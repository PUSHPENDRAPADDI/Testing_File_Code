### 3. Repository

```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using RepositoryLayer.IRepository;
using System.Collections.Generic;
using System.Linq;

namespace RepositoryLayer.Repository
{
    public class ClientProfileRepository : IRepository<ClientProfile>
    {
        private readonly ApplicationDbContext _context;

        public ClientProfileRepository(ApplicationDbContext context)
        {
            _context = context;
        }

        public void Delete(ClientProfile entity)
        {
            _context.ClientProfiles.Remove(entity);
            _context.SaveChanges();
        }

        public ClientProfile Get(int id)
        {
            return _context.ClientProfiles.SingleOrDefault(c => c.Id == id);
        }

        public IEnumerable<ClientProfile> GetAll()
        {
            return _context.ClientProfiles.ToList();
        }

        public void Insert(ClientProfile entity)
        {
            _context.ClientProfiles.Add(entity);
            _context.SaveChanges();
        }

        public void Update(ClientProfile entity)
        {
            _context.ClientProfiles.Update(entity);
            _context.SaveChanges();
        }

        public void SaveChanges()
        {
            _context.SaveChanges();
        }
    }
}
```

### 5. Controller

```csharp
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly ICustomService<ClientProfile> _clientProfileService;

        public ClientProfileController(ICustomService<ClientProfile> clientProfileService)
        {
            _clientProfileService = clientProfileService;
        }

        [HttpGet("{id}")]
        public IActionResult GetClientProfileById(int id)
        {
            var clientProfile = _clientProfileService.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }
            return Ok(clientProfile);
        }

        [HttpGet]
        public IActionResult GetAllClientProfiles()
        {
            var clientProfiles = _clientProfileService.GetAll();
            return Ok(clientProfiles);
        }

        [HttpPost]
        public IActionResult CreateClientProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client profile is null.");
            }
            _clientProfileService.Insert(clientProfile);
            return CreatedAtAction(nameof(GetClientProfileById), new { id = clientProfile.Id }, clientProfile);
        }

        [HttpPut("{id}")]
        public IActionResult UpdateClientProfile(int id, [FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null || clientProfile.Id != id)
            {
                return BadRequest("Client profile is null or ID mismatch.");
            }
            _clientProfileService.Update(clientProfile);
            return NoContent();
        }

        [HttpDelete("{id}")]
        public IActionResult DeleteClientProfile(int id)
        {
            var clientProfile = _clientProfileService.Get(id);
            if (clientProfile == null)
            {
                return NotFound();
            }
            _clientProfileService.Delete(clientProfile);
            return NoContent();
        }
    }
}
```
