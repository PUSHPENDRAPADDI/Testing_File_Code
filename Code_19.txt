### FastAPI Implementation

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional

app = FastAPI()

# Models
class Patient(BaseModel):
    id: int
    name: str
    contact_details: str
    medical_history: Optional[str] = None

class Appointment(BaseModel):
    id: int
    patient_id: int
    doctor_id: int
    date: str  # Use a proper datetime format in a real implementation

class MedicalRecord(BaseModel):
    patient_id: int
    record: str

class Prescription(BaseModel):
    id: int
    patient_id: int
    doctor_id: int
    details: str

class Invoice(BaseModel):
    id: int
    patient_id: int
    amount: float
    status: str

# In-memory storage (for demonstration purposes)
patients = []
appointments = []
medical_records = []
prescriptions = []
invoices = []

# Patient Management API
@app.post("/patients", response_model=Patient)
def register_patient(patient: Patient):
    patients.append(patient)
    return patient

@app.get("/patients/{id}", response_model=Patient)
def retrieve_patient(id: int):
    for patient in patients:
        if patient.id == id:
            return patient
    raise HTTPException(status_code=404, detail="Patient not found")

@app.put("/patients/{id}", response_model=Patient)
def update_patient(id: int, updated_patient: Patient):
    for idx, patient in enumerate(patients):
        if patient.id == id:
            patients[idx] = updated_patient
            return updated_patient
    raise HTTPException(status_code=404, detail="Patient not found")

@app.delete("/patients/{id}")
def deactivate_patient(id: int):
    for patient in patients:
        if patient.id == id:
            patients.remove(patient)
            return {"detail": "Patient profile deactivated"}
    raise HTTPException(status_code=404, detail="Patient not found")

# Appointment Scheduling API
@app.post("/appointments", response_model=Appointment)
def schedule_appointment(appointment: Appointment):
    appointments.append(appointment)
    return appointment

@app.put("/appointments/{id}", response_model=Appointment)
def modify_appointment(id: int, updated_appointment: Appointment):
    for idx, appointment in enumerate(appointments):
        if appointment.id == id:
            appointments[idx] = updated_appointment
            return updated_appointment
    raise HTTPException(status_code=404, detail="Appointment not found")

@app.get("/appointments")
def get_appointments():
    return appointments

# EMR API
@app.post("/patients/{patient_id}/records", response_model=MedicalRecord)
def add_medical_record(patient_id: int, record: MedicalRecord):
    medical_records.append(record)
    return record

@app.get("/patients/{patient_id}/records")
def get_medical_records(patient_id: int):
    return [record for record in medical_records if record.patient_id == patient_id]

# Prescription Management API
@app.post("/prescriptions", response_model=Prescription)
def generate_prescription(prescription: Prescription):
    prescriptions.append(prescription)
    return prescription

@app.get("/prescriptions/{id}")
def get_prescription(id: int):
    for prescription in prescriptions:
        if prescription.id == id:
            return prescription
    raise HTTPException(status_code=404, detail="Prescription not found")

# Billing & Insurance API
@app.post("/billing", response_model=Invoice)
def generate_invoice(invoice: Invoice):
    invoices.append(invoice)
    return invoice

@app.get("/billing/{id}")
def get_invoice(id: int):
    for invoice in invoices:
        if invoice.id == id:
            return invoice
    raise HTTPException(status_code=404, detail="Invoice not found")

# Run the application
if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

### Instructions to Run the Application

1. **Install FastAPI and Uvicorn** if you haven't already:

   ```bash
   pip install fastapi uvicorn
   ```

2. **Save the code** into a file named `main.py`.

3. **Run the FastAPI application**:

   ```bash
   uvicorn main:app --reload
   ```

4. **Access the API documentation** at `http://127.0.0.1:8000/docs`.
