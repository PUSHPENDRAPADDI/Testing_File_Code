### Complete ClientProfileController Implementation

Here's the complete implementation of `ClientProfileController` with data validation:

```csharp
using DomainLayer.Data;
using DomainLayer.Models;
using Microsoft.AspNetCore.Mvc;
using ServiceLayer.ICustomServices;

namespace OnionArchitectureInAspNetCore6WebAPI.Controllers
{
    [Route("api/[controller]")]
    [ApiController]
    public class ClientProfileController : ControllerBase
    {
        private readonly IClientManagementRepository _clientManagementRepository;

        public ClientProfileController(IClientManagementRepository clientManagementRepository)
        {
            _clientManagementRepository = clientManagementRepository;
        }

        [HttpPost("CreateClientProfile")]
        public IActionResult CreateClientProfile([FromBody] ClientProfile clientProfile)
        {
            if (clientProfile == null)
            {
                return BadRequest("Client Profile cannot be null.");
            }

            // Validate required fields
            if (string.IsNullOrWhiteSpace(clientProfile.FirstName) || 
                string.IsNullOrWhiteSpace(clientProfile.LastName) || 
                string.IsNullOrWhiteSpace(clientProfile.Email))
            {
                return BadRequest("First Name, Last Name, and Email are required.");
            }

            // Optionally validate the email format
            if (!IsValidEmail(clientProfile.Email))
            {
                return BadRequest("Invalid email format.");
            }

            _clientManagementRepository.Insert(clientProfile);
            return CreatedAtAction(nameof(CreateClientProfile), new { id = clientProfile.Id }, "Client Profile Created Successfully");
        }

        private bool IsValidEmail(string email)
        {
            try
            {
                var emailAddress = new System.Net.Mail.MailAddress(email);
                return emailAddress.Address == email;
            }
            catch
            {
                return false;
            }
        }
    }
}
```
