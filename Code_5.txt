### Repository Implementation

```csharp
using System.Collections.Generic;
using System.Threading.Tasks;
using Microsoft.EntityFrameworkCore;

namespace DomainLayer.Repositories
{
    public class ClientProfileRepository : IClientProfileRepository
    {
        private readonly DbContext _context;

        public ClientProfileRepository(DbContext context)
        {
            _context = context;
        }

        public async Task<ClientProfile> GetClientProfileByIdAsync(int id)
        {
            return await _context.Set<ClientProfile>().FindAsync(id);
        }

        public async Task<IEnumerable<ClientProfile>> GetAllClientProfilesAsync()
        {
            return await _context.Set<ClientProfile>().ToListAsync();
        }

        public async Task AddClientProfileAsync(ClientProfile clientProfile)
        {
            await _context.Set<ClientProfile>().AddAsync(clientProfile);
            await _context.SaveChangesAsync();
        }

        public async Task UpdateClientProfileAsync(ClientProfile clientProfile)
        {
            _context.Set<ClientProfile>().Update(clientProfile);
            await _context.SaveChangesAsync();
        }

        public async Task DeleteClientProfileAsync(int id)
        {
            var clientProfile = await GetClientProfileByIdAsync(id);
            if (clientProfile != null)
            {
                _context.Set<ClientProfile>().Remove(clientProfile);
                await _context.SaveChangesAsync();
            }
        }
    }
}
```
